** Namespace log =N1= - Simple namespace creation and deletion
- Create a namespace:
  #+BEGIN_SRC shell 
  kubectl create namespace rising
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  namespace/rising created
  #+end_src

- Get the namespace:
  #+BEGIN_SRC shell 
  kubectl get namespaces
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  NAME              STATUS   AGE
  default           Active   99d
  kube-flannel      Active   99d
  kube-node-lease   Active   99d
  kube-public       Active   99d
  kube-system       Active   99d
  limited           Active   99d
  my-namespace      Active   20d
  rising            Active   22h
  #+end_src
  
- Describe the namespace:
  #+BEGIN_SRC shell
  kubectl describe namespace rising
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  Name:         rising
  Labels:       kubernetes.io/metadata.name=rising
  Annotations:  <none>
  Status:       Active

  No resource quota.

  No LimitRange resource.
  #+end_src

- Delete the namespace:
  #+BEGIN_SRC shell
  kubectl delete namespace rising
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  namespace "rising" deleted
  #+end_src

** Namespace log =N2= - Namespace with resource quotas
- *Prerequisite*: recreate the =rising= namespace:
   #+BEGIN_SRC shell
   kubectl create namespace rising
   #+END_SRC

   #+RESULTS:
   #+begin_src shell
   Error from server (AlreadyExists): namespaces "rising" already exists
   #+end_src

- Create a ResourceQuota for the namespace:
   #+BEGIN_SRC shell
   cat <<EOF | kubectl apply -f -
   apiVersion: v1
   kind: ResourceQuota
   metadata:
     name: rising-quota 
   spec:
     hard:
       pods: "1"
       requests.cpu: "1"
       requests.memory: 1Gi
       limits.cpu: "2"
       limits.memory: 2Gi
   EOF
   #+END_SRC

   #+RESULTS:
   #+begin_src shell
   resourcequota/rising-quota created
   #+end_src

- Inspect the ResourceQuota:
  #+BEGIN_SRC shell
  kubectl -n rising describe resourcequota rising-quota
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  Name:            rising-quota
  Namespace:       rising
  Resource         Used  Hard
  --------         ----  ----
  limits.cpu       0     2
  limits.memory    0     2Gi
  pods             0     1
  requests.cpu     0     1
  requests.memory  0     1Gi
  #+end_src

- Create a Pod that will exceed the quota:
  #+BEGIN_SRC shell
  cat <<EOF | kubectl apply -f -
  apiVersion: v1
  kind: Pod
  metadata:
    name: over-quota-pod
    namespace: rising
  spec:
    containers:
    - name: over-quota-container
      image: nginx
      resources:
        requests:
          cpu: "1"
          memory: 1.5Gi
        limits:
          cpu: "2"
          memory: 2Gi
      command: ["sh", "-c", "echo Hello, Kubernetes! && sleep 3600"]
  EOF
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  Error from server (Forbidden): error when creating "STDIN": pods "over-quota-pod" is forbidden: exceeded quota: rising-quota, requested: requests.memory=1536Mi, used: requests.memory=0, limited: requests.memory=1Gi
  #+end_src

- Now create a Pod that will respect the quota:
  #+BEGIN_SRC shell
  cat <<EOF | kubectl apply -f -
  apiVersion: v1
  kind: Pod
  metadata:
    name: within-quota-pod
    namespace: rising
  spec:
    containers:
    - name: within-quota-container
      image: nginx
      resources:
        requests:
          cpu: "1"
          memory: 1Gi
        limits:
          cpu: "2"
          memory: 2Gi
      command: ["sh", "-c", "echo Hello, Kubernetes! && sleep 3600"]
  EOF
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  pod/within-quota-pod created
  #+end_src

- Run a further Pod, it will fail:
  #+BEGIN_SRC shell
  kubectl -n rising run --image=nginx nginx
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  Error from server (Forbidden): pods "nginx" is forbidden: failed quota: rising-quota: must specify limits.cpu for: nginx; limits.memory for: nginx; requests.cpu for: nginx; requests.memory for: nginx
  #+end_src

- Clean everything up, one pod will fail because it was never created:
  #+BEGIN_SRC shell
    kubectl -n rising delete pod over-quota-pod
    kubectl -n rising delete pod within-quota-pod
    kubectl delete resourcequota rising-quota
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  Error from server (NotFound): pods "over-quota-pod" not found
  pod "within-quota-pod" deleted
  resourcequota "rising-quota" deleted
  #+end_src

** Pod log =P2= - Image download
- *Prerequisite*: run the following helper script to wipe the image cache on all nodes:

  #+BEGIN_SRC shell :results type verbatim
    for node in $(kubectl get nodes -o jsonpath='{.items[*].metadata.name}'); do
      echo "Pruning images on $node"
      ssh -F ssh/cluster-setup $node "sudo crictl rmi --prune"
    done
  #+END_SRC
  
  #+RESULTS:
#+begin_src shell
Pruning images on kubeadm-master
Pruning images on kubeadm-worker1
Deleted: registry.k8s.io/pause:3.6

Deleted: docker.io/library/alpine:latest
Deleted: docker.io/library/nginx:latest
Pruning images on kubeadm-worker2
#+end_src

** ConfigMap =C1= - Create a ConfigMap and use it in a Pod, then delete it
- Create a ConfigMap:
  #+BEGIN_SRC shell
  cat <<EOF | kubectl apply -f -
  apiVersion: v1
  kind: ConfigMap
  metadata:
    name: my-config
    namespace: rising
  data:
    status: "decent" 
  EOF
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  configmap/my-config created
  #+end_src

- Get the configmap
  #+BEGIN_SRC shell
  kubectl -n rising get configmap my-config -o yaml
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  apiVersion: v1
  data:
    status: decent
  kind: ConfigMap
  metadata:
    annotations:
      kubectl.kubernetes.io/last-applied-configuration: |
        {"apiVersion":"v1","data":{"status":"decent"},"kind":"ConfigMap","metadata":{"annotations":{},"name":"my-config","namespace":"rising"}}
    creationTimestamp: "2024-05-29T08:24:20Z"
    name: my-config
    namespace: rising
    resourceVersion: "2345205"
    uid: 96f6a9d4-357e-4d79-a887-9ebd02b7be74
  #+end_src

- Deploy a Pod that uses the ConfigMap:
  #+BEGIN_SRC shell
  cat <<EOF | kubectl apply -f -
  apiVersion: v1
  kind: Pod
  metadata:
    name: config-pod
    namespace: rising
  spec:
    containers:
    - name: config-container
      image: alpine
      command: ["sh", "-c", 'echo Status is \$STATUS && sleep 3600']
      env:
      - name: STATUS
        valueFrom:
          configMapKeyRef:
            name: my-config
            key: status
  EOF
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  pod/config-pod created
  #+end_src

- Access the logs of the Pod:
  #+BEGIN_SRC shell
    kubectl wait --for=condition=Ready pod/config-pod -n rising
    kubectl -n rising logs config-pod -c config-container
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  pod/config-pod condition met
  Status is decent
  #+end_src

- Clean everything up:
  #+BEGIN_SRC shell
  kubectl -n rising delete pod config-pod
  kubectl -n rising delete configmap my-config
  #+END_SRC

  #+RESULTS:
  #+begin_src shell
  pod "config-pod" deleted
  configmap "my-config" deleted
  #+end_src

** Secret =S1= - Create a Secret and use it in a Pod, then delete it
- Create a secret
  #+BEGIN_SRC shell
  cat <<EOF | kubectl apply -f -
